rules_version = '2';

// turni iDEAli · Regole di sicurezza di Cloud Firestore
// Da incollare in: Console Firebase > Firestore Database > Regole > Pubblica.
//
// PRIMA DI PUBBLICARE: sostituisci UID_GESTIONE_1 con l'UID dell'account di gestione
// (Console Firebase > Authentication > Utenti > colonna "UID utente").
// Per un secondo account, sostituisci anche UID_GESTIONE_2; altrimenti lascialo com'è.

service cloud.firestore {
  match /databases/{database}/documents {

    function gestione() {
      return request.auth != null
        && request.auth.uid in ['UID_GESTIONE_1', 'UID_GESTIONE_2'];
    }

    function codice(h) {
      return get(/databases/$(database)/documents/codici/$(h)).data;
    }

    function raccolta(mese) {
      return get(/databases/$(database)/documents/raccolte/$(mese)).data;
    }

    // Formato compatto delle voci: 2026-12-25M1,2026-12-25P1,2026-12-31N2
    function vociValide(mese, voci) {
      return voci is string
        && voci.size() <= 1200
        && voci.matches('^(' + mese + '-(0[1-9]|[12][0-9]|3[01])[MPN][0-9]{1,2})(,' + mese + '-(0[1-9]|[12][0-9]|3[01])[MPN][0-9]{1,2})*$');
    }

    function rispostaValida(mese, h) {
      let d = request.resource.data;
      let c = codice(h);
      let r = raccolta(mese);
      return mese.matches('^20[0-9]{2}-(0[1-9]|1[0-2])$')
        && h.matches('^[0-9a-f]{64}$')
        && c.attivo == true
        && r.aperta == true
        && request.time < r.chiusura
        && d.keys().hasOnly(['figura', 'voci', 'disp', 'nota', 'inviata', 'v'])
        && d.keys().hasAll(['figura', 'voci', 'disp', 'nota', 'inviata', 'v'])
        && d.figura == c.figura
        && vociValide(mese, d.voci)
        && d.disp is int && d.disp >= 1 && d.disp <= d.voci.split(',').size()
        && d.nota is string && d.nota.size() <= 200
        && d.inviata == request.time
        && d.v == 1;
    }

    // Impostazioni generali (reparto, turni, valore casuale per le impronte): pubbliche, nessun dato personale.
    match /impostazioni/{doc} {
      allow read: if true;
      allow write: if gestione();
    }

    // Codici: sul server solo l'impronta (SHA-256), mai il codice in chiaro.
    // Chiunque può verificare un singolo codice; solo la gestione può elencarli o modificarli.
    match /codici/{h} {
      allow get: if true;
      allow list, write: if gestione();
    }

    // Raccolte mensili: visibili a tutti, modificabili solo dalla gestione.
    match /raccolte/{mese} {
      allow read: if true;
      allow write: if gestione();

      // Risposte: i colleghi possono solo scrivere la propria; nessuno, tranne la gestione, può leggerle.
      match /risposte/{h} {
        allow create, update: if rispostaValida(mese, h);
        allow read, delete: if gestione();
      }

      // Turni da coprire: solo gestione.
      match /riservato/{doc} {
        allow read, write: if gestione();
      }

      // Totale anonimo mostrato nella conferma.
      match /pubblico/{doc} {
        allow read: if true;
        allow write: if gestione();
      }
    }

    // Tutto il resto è chiuso.
    match /{percorso=**} {
      allow read, write: if false;
    }
  }
}
